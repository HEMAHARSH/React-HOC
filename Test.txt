from java.util import *
from weblogic.health import HealthState
import sys

# Connect to WebLogic
connect('{{ user_wlc }}', '{{ password_wlc }}',
        '{{ protocol_wlc }}://{{ server_wlc }}:{{ port_wlc }}')

domainRuntime()

servers = domainRuntimeService.getServerRuntimes()

output_data = "{\n"

for srv in servers:

    # ---- Server Name (GUI Name column) ----
    name = srv.getName()

    # ---- State column (RUNNING / SHUTDOWN etc.) ----
    state = srv.getState()

    # ---- Health column (OK / Warning / Critical etc.) ----
    healthObj = srv.getHealthState()
    health = HealthState.mapToString(healthObj.getState())

    # Remove HEALTH_ prefix to match GUI exactly
    health = health.replace("HEALTH_", "")

    print("Name   :", name)
    print("State  :", state)
    print("Health :", health)
    print("-------------------------------------")

    output_data += '  "' + name + '": {\n'
    output_data += '    "state": "' + state + '",\n'
    output_data += '    "health": "' + health + '"\n'
    output_data += '  },\n'

# Remove trailing comma
output_data = output_data.rstrip(',\n') + "\n}"

print("\nCurrent state of servers:\n")
print(output_data)

# Output file path
output_file_path = r'D:\apps\{{ app_path }}\applications\AdminServer\server_healthcheck_output.json'

output_file = open(output_file_path, 'w')
output_file.write(output_data)
output_file.close()

print('\nServer health JSON generated at:', output_file_path)

exit()
