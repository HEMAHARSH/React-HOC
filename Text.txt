---
- name: Check service status and generate report
  hosts: all
  gather_facts: no

  roles:
    - service

  tasks:

    - name: Generate HTML report
      template:
        src: "template.html.j2"
        dest: "service_status_report.html"
      vars:
        service_statuses: >-
          {{
            groups['all']
            | map('extract', hostvars, 'service_record')
            | list
          }}
      run_once: true
      delegate_to: localhost


---
- name: Check if service is running
  win_shell: "(Get-Service -Name '{{ service_name }}').Status"
  register: service_status
  ignore_errors: yes

- name: Set service status safely
  set_fact:
    final_status: >-
      {% if service_status is failed %}
      Not Running
      {% elif service_status.stdout | trim == "" %}
      Unknown
      {% else %}
      {{ service_status.stdout | trim }}
      {% endif %}

- name: Build service record
  set_fact:
    service_record:
      host: "{{ inventory_hostname }}"
      service_name: "{{ service_name }}"
      status: "{{ final_status }}"
      group: >-
        {% if 'CORE' in group_names %}
        CORE
        {% elif 'BATCH' in group_names %}
        BATCH
        {% elif 'ADAPTER' in group_names %}
        ADAPTER
        {% else %}
        UNKNOWN
        {% endif %}
