---
- name: Picasso Service Health Check
  hosts: all
  gather_facts: false

  tasks:

    - name: Check if service is running
      win_shell: "(Get-Service -Name '{{ service_name }}').Status"
      register: service_status

    - name: Build service record
      set_fact:
        service_record:
          host: "{{ inventory_hostname }}"
          service_name: "{{ service_name }}"
          status: "{{ service_status.stdout | default('Unknown') | trim }}"
          group: >-
            {% if 'CORE' in group_names %}
            CORE
            {% elif 'BATCH' in group_names %}
            BATCH
            {% elif 'ADAPTER' in group_names %}
            ADAPTER
            {% else %}
            UNKNOWN
            {% endif %}

    # 🔥 SAFE AGGREGATION (NO RACE CONDITION)
    - name: Generate HTML report
      template:
        src: "template.html.j2"
        dest: "service_status_report.html"
      vars:
        service_statuses: >-
          {{
            groups['all']
            | map('extract', hostvars, 'service_record')
            | list
          }}
      run_once: true
      delegate_to: localhost



<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Service Status Report</title>

<style>
body {
    font-family: Arial;
    margin: 20px;
}

h1 {
    text-align: center;
}

.section {
    margin-bottom: 30px;
}

table {
    width: 95%;
    border-collapse: collapse;
}

th, td {
    border: 1px solid #ccc;
    padding: 6px;
    text-align: left;
}

th {
    background-color: #f2f2f2;
}

.running {
    color: green;
    font-weight: bold;
}

.not-running {
    color: red;
    font-weight: bold;
}
</style>

</head>
<body>

<h1>Service Status Report</h1>

<!-- CORE -->
<div class="section">
<h2>CORE Services</h2>
<table>
<tr><th>Host</th><th>Service Name</th><th>Status</th></tr>

{% for entry in service_statuses if entry.group | trim == 'CORE' %}
<tr>
<td>{{ entry.host }}</td>
<td>{{ entry.service_name }}</td>
<td class="{% if entry.status | lower == 'running' %}running{% else %}not-running{% endif %}">
{{ entry.status }}
</td>
</tr>
{% endfor %}

</table>
</div>

<!-- BATCH -->
<div class="section">
<h2>BATCH Services</h2>
<table>
<tr><th>Host</th><th>Service Name</th><th>Status</th></tr>

{% for entry in service_statuses if entry.group | trim == 'BATCH' %}
<tr>
<td>{{ entry.host }}</td>
<td>{{ entry.service_name }}</td>
<td class="{% if entry.status | lower == 'running' %}running{% else %}not-running{% endif %}">
{{ entry.status }}
</td>
</tr>
{% endfor %}

</table>
</div>

<!-- ADAPTER -->
<div class="section">
<h2>ADAPTER Services</h2>
<table>
<tr><th>Host</th><th>Service Name</th><th>Status</th></tr>

{% for entry in service_statuses if entry.group | trim == 'ADAPTER' %}
<tr>
<td>{{ entry.host }}</td>
<td>{{ entry.service_name }}</td>
<td class="{% if entry.status | lower == 'running' %}running{% else %}not-running{% endif %}">
{{ entry.status }}
</td>
</tr>
{% endfor %}

</table>
</div>

</body>
</html>
