pipeline {

agent {
    label "Bnpp-Ansible-Cyberark"
}

environment {
    JDK_VERSION = "jdk8"   // adjust if needed
    ARTIFACTORY_URL = "https://artifactory.cib.echonet/artifactory"
    REPO = "picassonext-maven-local-dev"
    MAIL_LIST = "global_gecd_it_as_transversal_support@asia.bnpparibas.com"
}

stages {

    stage('Run All Environments') {
        steps {
            script {
                def environments = ['Batch-stg', 'Core-stg', 'Adapter-stg']

                for (envName in environments) {

                    stage("Run Ansible for ${envName}") {
                        wrap([$class: 'AnsiColorBuildWrapper', colorMapName: 'xterm']) {

                            withCredentials([
                                file(credentialsId: 'cyberarkCertFile', variable: 'cyberarkCert'),
                                file(credentialsId: 'cyberarkKeyFile', variable: 'cyberarkKey'),
                                file(credentialsId: 'cyberarkCAFile', variable: 'cyberarkCA'),
                                string(credentialsId: 'PICASSO_STG_LOGIN_MLC', variable: 'weblogic_login')
                            ]) {

                                echo "Running Ansible for environment: ${envName}"

                                sh """
                                ansible-playbook \
                                -e env=${envName} \
                                -e cyberark_cert=${cyberarkCert} \
                                -e cyberark_key=${cyberarkKey} \
                                -e cyberark_ca=${cyberarkCA} \
                                -e weblogic_stg=${weblogic_login} \
                                -e jdk_version=${JDK_VERSION} \
                                -i STAGING/Picasso_Server_HealthCheck_GUI/Inventories/hosts.ini \
                                STAGING/Picasso_Server_HealthCheck_GUI/playbook.yml
                                """
                            }
                        }
                    }
                }
            }
        }
    }

    stage('Merge All Environment Reports') {
        steps {
            script {
                def environments = ['Batch-stg', 'Core-stg', 'Adapter-stg']
                def combinedContent = ""

                for (envName in environments) {
                    def filePath = "${workspace}/server_report_${envName}.html"

                    if (fileExists(filePath)) {
                        def content = readFile(filePath)

                        combinedContent += """
                        <div style="border:2px solid #000;margin:20px;padding:20px;">
                            <h2 style="color:blue;">Environment: ${envName}</h2>
                            ${content}
                        </div>
                        """
                    } else {
                        combinedContent += """
                        <div style="color:red;">
                            <h2>Missing report for ${envName}</h2>
                        </div>
                        """
                    }
                }

                def finalHtml = """
                <html>
                <head>
                    <title>PICASSO Consolidated Report</title>
                </head>
                <body>
                    <h1 style="text-align:center;">PICASSO Health Check - Consolidated Report</h1>
                    ${combinedContent}
                </body>
                </html>
                """

                writeFile file: "${workspace}/final_report.html", text: finalHtml
            }
        }
    }

    stage('Send Consolidated Report') {
        steps {
            script {
                def reportPath = "${workspace}/final_report.html"

                if (fileExists(reportPath)) {
                    def htmlReport = readFile(reportPath)

                    emailext(
                        subject: "[PICASSO][STG]: Consolidated Health Check Report",
                        body: """
                        <html>
                        <body>
                        ${htmlReport}
                        </body>
                        </html>
                        """,
                        mimeType: 'text/html',
                        to: "${MAIL_LIST}"
                    )
                } else {
                    error "Final report not found!"
                }
            }
        }
    }
}

}
